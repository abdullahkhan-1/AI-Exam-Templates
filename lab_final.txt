# Class representing the grid environment
class GridEnvironment:
    def __init__(self, size):
        self.size = size   # Size of the grid (e.g., 10x10)

        # Create a grid filled with zeros
        self.grid = [[0 for _ in range(size)] for _ in range(size)]

        # Randomly generate the start position inside the grid
        self.start = (random.randint(0, size-1), random.randint(0, size-1))

        # Randomly generate the goal position inside the grid
        self.goal = (random.randint(0, size-1), random.randint(0, size-1))

    # Function to solve the CSP problem
    def solve_csp(self):

        # Create a CSP model
        model = cp_model.CpModel()

        # Extract x and y coordinates of start and goal positions
        x1, y1 = self.start # point A (0,3)(0,6)
        x2, y2 = self.goal

        # -------- Variables --------

        # Calculate horizontal distance between start and goal
        a = abs(x2 - x1)

        # Calculate vertical distance between start and goal
        b = abs(y2 - y1)

        # Create CSP variable for the hypotenuse (distance)
        # Range is from 0 to twice the grid size
        c = model.NewIntVar(0, self.size * 2, 'c')

        # Variables to store squared values
        a_sq = model.NewIntVar(0, self.size**2, 'a_sq') #a2
        b_sq = model.NewIntVar(0, self.size**2, 'b_sq')#3b2
        c_sq = model.NewIntVar(0, self.size**2, 'c_sq')

        # -------- Constraints --------

        # Define a_sq = a * a
        model.AddMultiplicationEquality(a_sq, [a, a])

        # Define b_sq = b * b
        model.AddMultiplicationEquality(b_sq, [b, b])

        # Define c_sq = c * c
        model.AddMultiplicationEquality(c_sq, [c, c])

        # Apply Pythagorean theorem: c² = a² + b²
        model.Add(c_sq == a_sq + b_sq)

        # -------- Solve the model --------

        # Create solver object
        solver = cp_model.CpSolver()

        # Solve the CSP model
        status = solver.Solve(model)

        # If a valid solution is found
        if status == cp_model.FEASIBLE or status == cp_model.OPTIMAL:

            # Return base, height, and calculated hypotenuse
            return a, b, solver.Value(c)

            # (This line will never execute because return is above)
            print(a, b, solver.Value(c))

        # If solver fails, calculate distance using normal math formula
        return a, b, math.sqrt(a**2 + b**2)


    # Function to display results
    def display(self):

        # Print start and goal positions
        print(f"Start Position: {self.start}")
        print(f"Goal Position: {self.goal}")

        # Solve the CSP problem and get distances
        a, b, c = self.solve_csp()

        # Display calculated distances
        print(f"Calculated Distances: Base={a}, Height={b}, Hypotenuse={c:.2f}")

def solve_pythagorean_with_obstacles(size, start, goal, obstacles):
    model = cp_model.CpModel()

    x1, y1 = start
    x2, y2 = goal
    
    # Calculate base and height
    a_val = abs(x2 - x1)
    b_val = abs(y2 - y1)

    # -------- Variables --------
    c = model.NewIntVar(0, size * 2, 'c')
    a_sq = model.NewIntVar(0, size**2, 'a_sq')
    b_sq = model.NewIntVar(0, size**2, 'b_sq')
    c_sq = model.NewIntVar(0, size**2, 'c_sq')

    # -------- Constraints --------
    
    # 1. Obstacle Check Logic
    # Agar koi obstacle start aur goal ke beech "Rectangle" mein hai, to path block hai.
    path_blocked = False
    for obs in obstacles:
        is_x_between = min(x1, x2) <= obs[0] <= max(x1, x2)
        is_y_between = min(y1, y2) <= obs[1] <= max(y1, y2)
        if is_x_between and is_y_between:
            path_blocked = True
            break

    if path_blocked:
        # Agar rasta block hai, to hum a ko domain se bahar (-1) kar dete hain taake solver fail ho jaye
        model.Add(c == -1) 
    else:
        # Define a_sq = a * a and b_sq = b * b
        model.AddMultiplicationEquality(a_sq, [a_val, a_val])
        model.AddMultiplicationEquality(b_sq, [b_val, b_val])
        
        # Define c_sq = c * c
        model.AddMultiplicationEquality(c_sq, [c, c])

        # Apply Pythagorean theorem: a² + b² = c²
        model.Add(c_sq == a_sq + b_sq)

    # -------- Solver --------
    solver = cp_model.CpSolver()
    status = solver.Solve(model)

    if status == cp_model.OPTIMAL or status == cp_model.FEASIBLE:
        return solver.Value(c), "Clear"
    else:
        # Agar block hai to math library se actual value return hogi magar status blocked ayega
        return math.sqrt(a_val**2 + b_val**2), "Blocked"

from ortools.sat.python import cp_model

def solve_timetable():
    model = cp_model.CpModel()
    courses = ['AI', 'DS', 'DB', 'COAL', 'OS']
    sizes = {'AI': 55, 'DS': 40, 'DB': 25, 'COAL': 35, 'OS': 20}
    
    # Variables
    slots = {c: model.NewIntVar(1, 3, f'slot_{c}') for c in courses}
    rooms = {c: model.NewIntVar(1, 2, f'room_{c}') for c in courses} # 1: R1(60), 2: R2(30)

    # 1. Room Capacity Constraint
    for c in courses:
        if sizes[c] > 30:
            model.Add(rooms[c] == 1)

    # 2. Student Overlap (Conflict) Constraints
    model.Add(slots['AI'] != slots['DS'])
    model.Add(slots['AI'] != slots['DB'])
    model.Add(slots['DS'] != slots['COAL'])
    model.Add(slots['DB'] != slots['OS'])

    # 3. Hard Rule: COAL in Slot 1 or 2
    model.AddAllowedSelections([slots['COAL']], [(1,), (2,)])

    # 4. Resource Constraint: If same slot, must be different room
    for i in range(len(courses)):
        for j in range(i + 1, len(courses)):
            c1, c2 = courses[i], courses[j]
            same_slot = model.NewBoolVar(f'{c1}_{c2}_same_slot')
            model.Add(slots[c1] == slots[c2]).OnlyEnforceIf(same_slot)
            model.Add(slots[c1] != slots[c2]).OnlyEnforceIf(same_slot.Not())
            model.Add(rooms[c1] != rooms[c2]).OnlyEnforceIf(same_slot)

    # 5. Extension: Soft Preference (AI in Slot 1)
    ai_in_s1 = model.NewBoolVar('ai_in_s1')
    model.Add(slots['AI'] == 1).OnlyEnforceIf(ai_in_s1)
    model.Minimize(ai_in_s1.Not())

    solver = cp_model.CpSolver()
    if solver.Solve(model) == cp_model.OPTIMAL:
        for c in courses:
            print(f"{c}: Slot {solver.Value(slots[c])}, Room R{solver.Value(rooms[c])}")
    else:
        print("No solution found.")

solve_timetable()

from ortools.sat.python import cp_model

def solve_traffic():
    model = cp_model.CpModel()
    nodes = ['A', 'B', 'C', 'D']
    # 0 = NorthSouth (NS), 1 = EastWest (EW)
    vars = {n: model.NewIntVar(0, 1, n) for n in nodes}

    # Unary Constraint: D cannot use EW
    model.Add(vars['D'] == 0)

    # Binary Adjacency Constraints
    edges = [('A', 'B'), ('A', 'C'), ('B', 'D'), ('C', 'D')]
    for u, v in edges:
        model.Add(vars[u] != vars[v])

    class SolutionPrinter(cp_model.CpSolverSolutionCallback):
        def __init__(self, vars):
            cp_model.CpSolverSolutionCallback.__init__(self)
            self.__vars = vars
            self.count = 0
        def on_solution_callback(self):
            self.count += 1
            mapping = {0: 'NS', 1: 'EW'}
            res = {n: mapping[self.Value(self.__vars[n])] for n in nodes}
            print(f"Assignment {self.count}: {res}")

    solver = cp_model.CpSolver()
    printer = SolutionPrinter(vars)
    print("--- Traffic Light Assignments ---")
    solver.SearchForAllSolutions(model, printer)

solve_traffic()

from ortools.sat.python import cp_model

def solve_n_queens():
    model = cp_model.CpModel()
    n = 8
    # queens[i] = row index for queen in column i
    queens = [model.NewIntVar(0, n - 1, f'q{i}') for i in range(n)]

    model.AddAllDifferent(queens)
    model.AddAllDifferent([queens[i] + i for i in range(n)]) # Upper diagonals
    model.AddAllDifferent([queens[i] - i for i in range(n)]) # Lower diagonals

    class NQueensPrinter(cp_model.CpSolverSolutionCallback):
        def __init__(self, queens):
            cp_model.CpSolverSolutionCallback.__init__(self)
            self.queens = queens
            self.count = 0
        def on_solution_callback(self):
            self.count += 1
            if self.count <= 3:
                print(f"Board {self.count}:")
                for row in range(n):
                    line = ""
                    for col in range(n):
                        if self.Value(self.queens[col]) == row: line += " Q "
                        else: line += " . "
                    print(line)
                print("-" * 25)

    solver = cp_model.CpSolver()
    printer = NQueensPrinter(queens)
    print("--- N-Queens (8 Cameras) ---")
    solver.SearchForAllSolutions(model, printer)
    print(f"Total Solutions Found: {printer.count}")

solve_n_queens()

def solve_relief(budget_max=150, med_min=8):
    model = cp_model.CpModel()
    F = model.NewIntVar(0, 100, 'Food')
    W = model.NewIntVar(0, 100, 'Water')
    M = model.NewIntVar(0, 100, 'Medicine')

    # Constraints
    model.Add(4*F + 3*W + 5*M <= 180)  # Volume
    model.Add(6*F + 2*W + 4*M <= 200)  # Weight
    model.Add(3*F + 5*W + 2*M <= budget_max)
    model.Add(W >= 10)
    model.Add(M >= med_min)

    # Objective: Maximize impact
    model.Maximize(7*F + 6*W + 9*M)

    solver = cp_model.CpSolver()
    status = solver.Solve(model)
    
    if status == cp_model.OPTIMAL:
        return {
            'Impact': solver.ObjectiveValue(),
            'Units': (solver.Value(F), solver.Value(W), solver.Value(M)),
            'Stats': f"Time: {solver.WallTime():.4f}s"
        }
    return None

def solve_warehouse_diagonal(grid_size, start, goal, obstacles):
    model = cp_model.CpModel()

    x1, y1 = start
    x2, y2 = goal
    
    # Calculate base (horizontal) and height (vertical) differences
    a_val = abs(x2 - x1)
    b_val = abs(y2 - y1)

    # -------- Variables --------
    # c represents the diagonal distance 
    # Hum sqrt(2) ka factor pattern ke hisab se simple integers me handle karte hain
    c_val_int = int(math.sqrt(a_val**2 + b_val**2))
    
    c = model.NewIntVar(0, grid_size * 2, 'diagonal_dist')
    a_sq = model.NewIntVar(0, grid_size**2, 'a_sq')
    b_sq = model.NewIntVar(0, grid_size**2, 'b_sq')
    c_sq = model.NewIntVar(0, grid_size**2, 'c_sq')

    # -------- Constraints --------
    
    # 1. Obstacle Check Logic (Rectangular zone checking)
    # Agar start aur goal ke bounding box mein koi obstacle hai, toh path valid nahi.
    path_blocked = False
    for obs in obstacles:
        is_x_between = min(x1, x2) <= obs[0] <= max(x1, x2)
        is_y_between = min(y1, y2) <= obs[1] <= max(y1, y2)
        if is_x_between and is_y_between:
            path_blocked = True
            break

    # 2. Parity Check for Diagonal Movement
    # Warehouse robot sirf diagonal move kar sakta hai, 
    # iska matlab start aur target ka parity (sum of coordinates) same hona chahiye.
    if (x1 + y1) % 2 != (x2 + y2) % 2:
        path_blocked = True

    if path_blocked:
        # Domain constraint fail karne ke liye
        model.Add(c == -1) 
    else:
        # a_sq = a * a aur b_sq = b * b set karna
        model.AddMultiplicationEquality(a_sq, [a_val, a_val])
        model.AddMultiplicationEquality(b_sq, [b_val, b_val])
        
        # c_sq = c * c
        model.AddMultiplicationEquality(c_sq, [c, c])

        # Pythagorean theorem: a² + b² = c²
        model.Add(c_sq == a_sq + b_sq)

    # -------- Solver --------
    solver = cp_model.CpSolver()
    status = solver.Solve(model)

    if status == cp_model.OPTIMAL or status == cp_model.FEASIBLE:
        # Diagonal cost sqrt(2) hoti hai, pattern ke mutabiq distance return karein
        actual_dist = math.sqrt(a_val**2 + b_val**2)
        return actual_dist, "Valid Diagonal Path"
    else:
        # Agar solve nahi hua toh blocked status
        return math.sqrt(a_val**2 + b_val**2), "Path Blocked/Invalid Parity"

import math

def check_winner(board):
    for i in range(3):
        if board[i][0] == board[i][1] == board[i][2] != ' ':
            return board[i][0]
        if board[0][i] == board[1][i] == board[2][i] != ' ':
            return board[0][i]
    if board[0][0] == board[1][1] == board[2][2] != ' ':
        return board[0][0]
    if board[0][2] == board[1][1] == board[2][0] != ' ':
        return board[0][2]
    if all(cell != ' ' for row in board for cell in row):
        return 'Draw'
    return None

def minimax(board, is_maximizing):
    res = check_winner(board)
    if res == 'X': return 1
    if res == 'O': return -1
    if res == 'Draw': return 0

    if is_maximizing:
        best_score = -math.inf
        for r in range(3):
            for c in range(3):
                if board[r][c] == ' ':
                    board[r][c] = 'X'
                    score = minimax(board, False)
                    board[r][c] = ' '
                    best_score = max(score, best_score)
        return best_score
    else:
        best_score = math.inf
        for r in range(3):
            for c in range(3):
                if board[r][c] == ' ':
                    board[r][c] = 'O'
                    score = minimax(board, True)
                    board[r][c] = ' '
                    best_score = min(score, best_score)
        return best_score

def minmax(node, depth, maximizing_player=True):
    if depth == 0 or not node.children:
        return node.value

    if maximizing_player:
        value = -math.inf
        for child in node.children:
            child_value = minmax(child, depth-1, False)
            value = max(value, child_value) # max(0,4)
        node.minmax_value = value
        return value
    else:
        value = math.inf
        for child in node.children:
            child_value = minmax(child, depth-1, True)
            value = min(value, child_value)
        node.minmax_value = value
        return value

        self.value = value
        self.children = []

def alpha_beta(node, depth, alpha, beta, maximizing_player=True):
    if depth == 0 or not node.children:
        return node.value

    if maximizing_player:
        value = -math.inf
        for child in node.children:
            value = max(value, alpha_beta(child, depth-1, alpha, beta, False))
            alpha = max(alpha, value)
            if beta <= alpha: # alpha >= beta
                print("Pruned node:", node)
                break
        node.minmax_value = value
        return value
    else:
        value = math.inf
        for child in node.children:
            value = min(value, alpha_beta(child, depth-1, alpha, beta, True))
            beta = min(beta, value)
            if beta < alpha:
                print("Pruned node:", child.value)
                break
        node.minmax_value = value
        return value

#MARKOV MODEL:
import numpy as np

# Define the states and transition matrix
states = ["Red", "Blue"]
transition_matrix = np.array([[0.5, 0.5],  # From Red -> Red or Blue
                              [0.5, 0.5]]) # From Blue -> Red or Blue

# Function to simulate the Markov process
def simulate_markov_process(initial_state, num_steps):
    current_state = initial_state
    state_sequence = [current_state]

    for _ in range(num_steps):
        if current_state == "Red":
            next_state = np.random.choice(states, p=transition_matrix[0])
        else:
            next_state = np.random.choice(states, p=transition_matrix[1])

        state_sequence.append(next_state)
        current_state = next_state

    return state_sequence

# Simulate the process starting from "Red" and for 10 steps
initial_state = "Red"
num_steps = 10
state_sequence = simulate_markov_process(initial_state, num_steps)

# Output the sequence of states
print(f"State sequence for {num_steps} steps starting from {initial_state}:")
print(" -> ".join(state_sequence))

import numpy as np

states = ['Sunny', 'Cloudy', 'Rainy']
state_map = {0: 'Sunny', 1: 'Cloudy', 2: 'Rainy'}

transition_matrix = [
    [0.6, 0.3, 0.1],
    [0.4, 0.4, 0.2],
    [0.2, 0.3, 0.5]
]

def simulate_weather(days, start_state):
    current_state_idx = states.index(start_state)
    sequence = [start_state]

    for _ in range(days - 1):
        next_state_idx = np.random.choice(
            [0, 1, 2],
            p=transition_matrix[current_state_idx]
        )
        sequence.append(state_map[next_state_idx])
        current_state_idx = next_state_idx

    return sequence

simulation_10_days = simulate_weather(10, 'Sunny')
print(f"Weather for the next 10 days: {simulation_10_days}")

trials = 100000
at_least_3_rainy = 0

for _ in range(trials):
    result = simulate_weather(10, 'Sunny')
    rainy_days = result.count('Rainy')
    if rainy_days >= 3:
        at_least_3_rainy += 1

final_probability = at_least_3_rainy / trials
print(f"Probability of at least 3 rainy days: {final_probability:.4f}")
